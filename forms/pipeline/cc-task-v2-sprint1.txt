    private List<FormProject> _projects = new();
    private bool _loading = true;

    protected override async Task OnInitializedAsync()
    {
        await LoadProjects();
    }

    private async Task LoadProjects()
    {
        _loading = true;
        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();
            _projects = await db.FormProjects
                .Include(p => p.Documents)
                .OrderByDescending(p => p.CreatedAt)
                .ToListAsync();
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Failed to load projects: {ex.Message}", Severity.Error);
        }
        finally
        {
            _loading = false;
        }
    }

    private async Task OpenNewProjectDialog()
    {
        var dialog = await DialogService.ShowAsync<ProjectDialog>("New Project");
        var result = await dialog.Result;
        if (!result.Canceled && result.Data is ProjectDialogResult data)
        {
            await CreateProject(data);
        }
    }

    private async Task CreateProject(ProjectDialogResult data)
    {
        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();
            var project = new FormProject
            {
                Name = data.Name,
                Vertical = data.Vertical,
                Description = data.Description,
                Status = "draft",
                CreatedAt = DateTime.UtcNow,
                UpdatedAt = DateTime.UtcNow
            };
            db.FormProjects.Add(project);
            await db.SaveChangesAsync();
            Snackbar.Add($"Project '{data.Name}' created!", Severity.Success);
            await LoadProjects();
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Failed to create project: {ex.Message}", Severity.Error);
        }
    }

    private async Task DeleteProject(FormProject project)
    {
        var confirm = await DialogService.ShowMessageBox(
            "Delete Project",
            $"Delete '{project.Name}'? This will unlink all associated documents.",
            yesText: "Delete", noText: "Cancel");
        if (confirm != true) return;

        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();
            var p = await db.FormProjects.FindAsync(project.Id);
            if (p != null)
            {
                db.FormProjects.Remove(p);
                await db.SaveChangesAsync();
                Snackbar.Add($"Project '{project.Name}' deleted.", Severity.Info);
                await LoadProjects();
            }
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Failed to delete: {ex.Message}", Severity.Error);
        }
    }

    [Parameter] public int ProjectId { get; set; }

    private FormProject? _project;
    private List<FormLibrary> _documents = new();
    private List<QuestionSet> _questionSets = new();
    private bool _loading = true;
    private bool _uploading = false;

    protected override async Task OnInitializedAsync()
    {
        await LoadProject();
    }

    private async Task LoadProject()
    {
        _loading = true;
        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();
            _project = await db.FormProjects
                .Include(p => p.Documents)
                .Include(p => p.QuestionSets)
                    .ThenInclude(q => q.Fields)
                .FirstOrDefaultAsync(p => p.Id == ProjectId);

            if (_project != null)
            {
                _documents = _project.Documents.OrderByDescending(d => d.CreatedAt).ToList();
                _questionSets = _project.QuestionSets.OrderByDescending(q => q.CreatedAt).ToList();
            }
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Failed to load project: {ex.Message}", Severity.Error);
        }
        finally
        {
            _loading = false;
        }
    }

    private async Task HandleFileUpload(InputFileChangeEventArgs e)
    {
        _uploading = true;
        StateHasChanged();
        try
        {
            var content = new MultipartFormDataContent();
            foreach (var file in e.GetMultipleFiles(10))
            {
                var stream = file.OpenReadStream(50 * 1024 * 1024);
                var sc = new StreamContent(stream);
                sc.Headers.ContentType = new System.Net.Http.Headers.MediaTypeHeaderValue("application/pdf");
                content.Add(sc, "files", file.Name);
            }
            content.Add(new StringContent("Unknown"), "carrierName");
            content.Add(new StringContent("Carrier"), "formType");
            content.Add(new StringContent(ProjectId.ToString()), "projectId");

            var response = await Http.PostAsync("/api/forms/upload", content);
            if (response.IsSuccessStatusCode)
            {
                Snackbar.Add("Files uploaded successfully!", Severity.Success);
                await LoadProject();
            }
            else
            {
                Snackbar.Add("Upload failed.", Severity.Error);
            }
        }
        catch (Exception ex)
        {
            Snackbar.Add($"Upload error: {ex.Message}", Severity.Error);
        }
        finally
        {
            _uploading = false;
            StateHasChanged();
        }
    }

    private async Task ExtractAllPending()
    {
        var pendingDocs = _documents.Where(d => d.Status == "Queued").ToList();
        foreach (var doc in pendingDocs)
        {
            await Http.PostAsync($"/api/forms/{doc.Id}/resubmit", null);
        }
        Snackbar.Add($"Queued {pendingDocs.Count} document(s) for extraction.", Severity.Info);
        await LoadProject();
    }

    private async Task ReExtract(int docId)
    {
        await Http.PostAsync($"/api/forms/{docId}/resubmit", null);
        Snackbar.Add("Re-extraction queued.", Severity.Info);
        await LoadProject();
    }

    private string GetVerticalEmoji(string vertical) => vertical switch

        _ => "📋"
    };

    private Color GetStatusColor(string status) => status switch
    {
        "draft" => Color.Default,
        "extracting" => Color.Warning,
        "extracted" => Color.Info,
        "building" => Color.Primary,
        "complete" => Color.Success,
        _ => Color.Default
    };

    private Color GetDocStatusColor(string status) => status switch
    {
        "Queued" => Color.Default,
        "Processing" => Color.Warning,
        "Draft" => Color.Info,
        "Reviewed" => Color.Primary,
        "Approved" => Color.Success,
        "Error" => Color.Error,
        _ => Color.Default
    };
